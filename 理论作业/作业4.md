### 1. 读者写者问题

```c
int readcount = 0;      // 正在读的读者数量
int writecount = 0;     // 正在等待或准准备写的写者数量
Semaphore r_mutex = 1;  // 保护 readcount
Semaphore w_mutex = 1;  // 保护 writecount
Semaphore read_lock = 1;// 控制读者进入的栅栏（有写者时堵住读者）
Semaphore resource = 1; // 资源访问互斥锁（读写互斥、写写互斥）

Writer() {
    P(w_mutex);
    if (++writecount == 1) P(read_lock); // 第一个写者到达，封锁读者流
    V(w_mutex);

    P(resource);        // 互斥访问资源
    // --- 执行写操作 ---
    V(resource);

    P(w_mutex);
    if (--writecount == 0) V(read_lock); // 最后一个写者离开，释放读者流
    V(w_mutex);
}

Reader() {
    P(read_lock);       // 检查是否有写者在等待（如有，则在此排队）
    P(r_mutex);
    if (++readcount == 1) P(resource);   // 第一个读者占用资源锁
    V(r_mutex);
    V(read_lock);       // 释放栅栏，允许后续读者进入

    // --- 执行读操作 ---

    P(r_mutex);
    if (--readcount == 0) V(resource);   // 最后一个读者释放资源锁
    V(r_mutex);
}
```


### 2. 寿司店问题

```c
int count = 0;          // 当前就坐人数
Semaphore mutex = 1;    // 保护 count 变量的互斥信号量
Semaphore empty = 5;    // 可用座位信号量
Semaphore waitAll = 0;  // 用于控制“全走完才能再进”的同步信号量

Customer() {
    P(empty);           // 申请座位
    P(mutex);
    count++;
    V(mutex);

    // --- 就餐中 ---

    P(mutex);
    count--;
    if (count == 0) {   // 如果我是最后一个走的
        for(int i=0; i<5; i++) V(empty); // 释放5个位置给下一拨
    }
    V(mutex);
}
```

### 3. 进门问题

```c
int count = 0;
Semaphore mutex = 1;    // 保护 count
Semaphore barrier = 0;  // 等待所有人刷卡
Semaphore door = 1;     // 保证一个一个过大门

Employee() {
    // 阶段1：刷卡同步
    P(mutex);
    count++;
    if (count == 5) V(barrier); // 第5个人唤醒第一位开始进入
    V(mutex);
    
    P(barrier);         // 等待信号
    V(barrier);         // 传递信号给下一个同事

    // 阶段2：一个一个过大门
    P(door);
    // --- 通过大门 ---
    V(door);

    // 阶段3：最后一人关门
    P(mutex);
    count--;
    if (count == 0) {
        // 关门操作
    }
    V(mutex);
}
```

### 4. 搜索-插入-删除问题

```c
int scount = 0;         // 搜索者数量
Semaphore r_mutex = 1;  // 保护 scount
Semaphore i_mutex = 1;  // 插入者之间的互斥
Semaphore s_d_lock = 1; // 搜索与删除的互斥
Semaphore i_d_lock = 1; // 插入与删除的互斥

Searcher() {
    P(r_mutex);
    if (++scount == 1) P(s_d_lock); // 第一个搜索者阻止删除
    V(r_mutex);
    
    // --- 执行搜索 ---
    
    P(r_mutex);
    if (--scount == 0) V(s_d_lock); // 最后一个搜索者释放删除
    V(r_mutex);
}

Inserter() {
    P(i_mutex);      // 插入者之间互斥
    P(i_d_lock);     // 阻止删除
    
    // --- 执行插入 ---
    
    V(i_d_lock);
    V(i_mutex);
}

Deleter() {
    P(s_d_lock);     // 确保没有人在搜索
    P(i_d_lock);     // 确保没有人在插入
    
    // --- 执行删除 ---
    
    V(i_d_lock);
    V(s_d_lock);
}
```

### 5. 公园购票问题

```c
int readcount = 0;      // 当前正在查询的人数
int quota = N;          // 剩余票数
Semaphore rmutex = 1;   // 保护 readcount 变量的互斥访问
Semaphore mutex = 1;    // 保护 quota 变量（写操作）的互斥访问
Semaphore in = 1;       // 保证购票者和查询者按到达顺序排队的“大门”信号量

// --- 购票者 B (相当于写者) ---
buy() {
    // 第一次检查：如果已经没票，直接返回
    if (quota == 0) {
        print("sold out!");
        return;
    }

    P(in);          // 步入排队区，保证顺序
    P(mutex);       // 申请对票数 quota 的独占访问权

    // 第二次检查：在获取锁后再次确认余票（防止在等待锁时票被买光）
    if (quota == 0) {
        print("sold out!");
        V(mutex);   // 注意释放锁
        V(in);
        return;
    } 
    else {
        quota--;    // 执行写操作：扣减票数
    }

    V(mutex);       // 释放票数锁
    V(in);          // 离开排队区
}

// --- 查询者 Q (相当于读者) ---
query() {
    P(in);          // 步入排队区，按序等待
    P(rmutex);      // 保护 readcount 的修改

    if (++readcount == 1) {
        P(mutex);   // 第一个查询者要占住 quota 锁，防止购票者修改
    }

    V(rmutex);
    V(in);          // 释放排队锁，允许后续的人（包括购票者）排队

    read the "quota"; // 执行读操作：查看余票

    P(rmutex);
    if (--readcount == 0) {
        V(mutex);   // 最后一个查询者离开，释放 quota 锁
    }
    V(rmutex);
}
```
